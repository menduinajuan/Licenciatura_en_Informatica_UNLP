			.data
CONTROL:		.word32 0x10000
DATA:			.word32 0x10008
color_fondo:		.word32 0x00FFFFFF							; Blanco
X:			.word 23, 10, 40, 5, 30							; Posición X de cada pelota
Y:			.word 1, 20, 35, 45, 10							; Posición Y de cada pelota
DX:			.word 1, -1, 1, -1, 1							; Dirección X
DY:			.word 1, 1, -1, -1, 1							; Dirección Y
COLOR:			.word 0x00FF0000, 0x000000FF, 0x0000FF00, 0x00FF00FF, 0x0000FFFF	; azul, rojo, verde, magenta, amarillo

			.text
			lwu $s6, CONTROL($zero)
			lwu $s7, DATA($zero)
			lwu $v1, color_fondo($zero)
			daddi $s4, $zero, 5							; Comando para dibujar un punto

LOOP:			daddi $t2, $zero, 0							; Desplazamiento en las tablas (0, 8, 16, ...)
			daddi $t3, $zero, 5							; Cantidad de pelotas

PELOTA:			ld $s0, X($t2)								; Carga los datos de la pelota actual
			ld $s1, Y($t2)
			ld $s2, DX($t2)
			ld $s3, DY($t2)
			ld $v0, COLOR($t2)
			sw $v1, 0($s7)								; Borra la pelota
			sb $s0, 4($s7)
			sb $s1, 5($s7)
			sd $s4, 0($s6)
			dadd $s0, $s0, $s2							; Mueve la pelota
			dadd $s1, $s1, $s3
			daddi $t1, $zero, 48							; Rebotes (igual que el original)
			slt $t0, $t1, $s0
			dsll $t0, $t0, 1
			dsub $s2, $s2, $t0
			slt $t0, $t1, $s1
			dsll $t0, $t0, 1
			dsub $s3, $s3, $t0
			slti $t0, $s0, 1
			dsll $t0, $t0, 1
			dadd $s2, $s2, $t0
			slti $t0, $s1, 1
			dsll $t0, $t0, 1
			dadd $s3, $s3, $t0
			sw $v0, 0($s7)								; Dibuja la pelota
			sb $s0, 4($s7)
			sb $s1, 5($s7)
			sd $s4, 0($s6)
			sd $s0, X($t2)								; Guarda los valores nuevos
			sd $s1, Y($t2)
			sd $s2, DX($t2)
			sd $s3, DY($t2)
			daddi $t2, $t2, 8							; Pasa a la siguiente pelota
			daddi $t3, $t3, -1
			bnez $t3, PELOTA
			daddi $t0, $zero, 500							; Demora

DEMORA:			daddi $t0, $t0, -1
			bnez $t0, DEMORA
			j LOOP
